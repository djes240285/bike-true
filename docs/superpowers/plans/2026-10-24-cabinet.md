# Cabinet Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Человек входит через Google, Яндекс или Telegram и хранит мерки, велосипеды и снимки углов; разбор фото по-прежнему считается в браузере.

**Architecture:** Чистые правила живут в `src/lib` и покрыты `bun test`. HTTP-сервер Hono в `api/` пишет в MySQL 8 и импортирует эти правила. Страницы кабинета — React-островки Astro, они ходят на API с куку `bt_session` и заголовком `X-BikeTrue: 1`. Заявка визита остаётся функцией Netlify.

**Tech Stack:** Astro, React, Bun, Hono, Drizzle ORM, mysql2, MySQL 8.

## Global Constraints

- Сайт остаётся Astro на Netlify. API — отдельный процесс Bun, каталог `api/`, порт 8787.
- База — MySQL 8, база `biketrue`. Клиент MySQL 5.7 не использовать.
- Пока сайт закрыт паролем, API только локально, на прод не выкладывать.
- Фото и кадры на сервер не отправлять. Локальные углы и кадры в браузере при сохранении не стирать.
- Коридоры в `src/lib/advice.ts` не менять. Посадка в снимке — подпись, числа на экране разбора от неё не зависят.
- `inseam_cm` — расстояние от верхней грани книги до пола в обуви для катания. Поля высоты подошвы нет.
- Кука `bt_session`: `HttpOnly`, `Secure`, `SameSite=None`, `Path=/`, 30 дней. В базе sha256 токена.
- Изменяющий запрос без заголовка `X-BikeTrue: 1` отвечает 403 `{ "ok": false, "error": "forbidden" }`.
- CORS только на `WEB_ORIGIN`, с учётными данными.
- Ошибка API: `{ "ok": false, "error": "<код>" }`. Успех мутации: `{ "ok": true }`.
- Коды: `session`, `forbidden`, `state`, `provider`, `telegram`, `taken`, `name`, `rider`, `height`, `inseam`, `fit`, `bike`, `limit`, `angles`, `photo`.
- Не больше 20 велосипедов и 100 снимков на человека.
- Роль `fitter` в интерфейсе не выдаётся.
- Секреты в репозиторий не коммитить. В `.env.example` только имена.
- Тесты правил: `bun test src/lib`. Живой MySQL в прогон не входит.
- Русский без префикса, английский под `/en`. Строки только в `src/i18n.ts`.
- Вне спеки не делать: разные коридоры, подбор до покупки, формулу седла, метки, видео, оплату, слияние двух учёток, перенос визита в MySQL.

`src/lib/account.ts`:

```ts
import { safeNext } from './gate';
import { isFit, isRider, type FitId, type RiderId } from './fits';
import { hasPhotoField } from './visit';

export const BIKE_LIMIT = 20;
export const READING_LIMIT = 100;

export function overLimit(count: number, limit: number): boolean {
  return count >= limit;
}

export function cabinetNext(raw: string | null | undefined): string {
  const next = safeNext(raw);
  return next === '/' ? '/account' : next;
}

export function parseDisplayName(value: unknown): { ok: true; name: string } | { ok: false; error: 'name' } {
  const name = typeof value === 'string' ? value.trim() : '';
  if (!name || name.length > 80) return { ok: false, error: 'name' };
  return { ok: true, name };
}

function emptyInt(value: unknown): unknown {
  return value === '' ? null : value;
}

function optionalInt(value: unknown, min: number, max: number): number | null | undefined {
  const raw = emptyInt(value);
  if (raw === null) return null;
  if (typeof raw === 'number' && Number.isInteger(raw) && raw >= min && raw <= max) return raw;
  return undefined;
}

export function parseProfile(body: unknown) {
  if (hasPhotoField(body)) return { ok: false as const, error: 'photo' as const };
  if (!body || typeof body !== 'object') return { ok: false as const, error: 'rider' as const };
  const record = body as Record<string, unknown>;
  if (record.rider !== null && !isRider(record.rider)) return { ok: false as const, error: 'rider' as const };
  const heightCm = optionalInt(record.height_cm, 90, 220);
  if (heightCm === undefined) return { ok: false as const, error: 'height' as const };
  const inseamCm = optionalInt(record.inseam_cm, 40, 110);
  if (inseamCm === undefined) return { ok: false as const, error: 'inseam' as const };
  if (heightCm !== null && inseamCm !== null && inseamCm >= heightCm) return { ok: false as const, error: 'inseam' as const };
  return { ok: true as const, profile: { rider: record.rider as RiderId | null, heightCm, inseamCm } };
}

export function parseBike(body: unknown) {
  if (hasPhotoField(body)) return { ok: false as const, error: 'photo' as const };
  if (!body || typeof body !== 'object') return { ok: false as const, error: 'name' as const };
  const record = body as Record<string, unknown>;
  const name = typeof record.name === 'string' ? record.name.trim() : '';
  if (!name || name.length > 40) return { ok: false as const, error: 'name' as const };
  if (!isFit(record.fit)) return { ok: false as const, error: 'fit' as const };
  const crankMm = optionalInt(record.crank_mm, 120, 180);
  const saddleMm = optionalInt(record.saddle_mm, 400, 1000);
  const stemMm = optionalInt(record.stem_mm, 30, 150);
  const dropMm = optionalInt(record.drop_mm, -150, 200);
  if (crankMm === undefined || saddleMm === undefined || stemMm === undefined || dropMm === undefined) {
    return { ok: false as const, error: 'bike' as const };
  }
  return { ok: true as const, bike: { name, fit: record.fit as FitId, crankMm, saddleMm, stemMm, dropMm } };
}

export function parseReading(body: unknown) {
  if (hasPhotoField(body)) return { ok: false as const, error: 'photo' as const };
  if (!body || typeof body !== 'object') return { ok: false as const, error: 'fit' as const };
  const record = body as Record<string, unknown>;
  if (!isFit(record.fit)) return { ok: false as const, error: 'fit' as const };
  const bikeRaw = emptyInt(record.bike_id);
  let bikeId: number | null = null;
  if (bikeRaw !== null && bikeRaw !== undefined) {
    if (typeof bikeRaw !== 'number' || !Number.isInteger(bikeRaw) || bikeRaw < 1) return { ok: false as const, error: 'bike' as const };
    bikeId = bikeRaw;
  }
  const angles = ['knee', 'back', 'arm', 'hip'] as const;
  const reading = { fit: record.fit as FitId, bikeId, knee: null as number | null, back: null as number | null, arm: null as number | null, hip: null as number | null, hadTop: record.had_top === true };
  for (const key of angles) {
    const value = record[key];
    if (value === null || value === undefined) continue;
    if (typeof value !== 'number' || !Number.isInteger(value)) return { ok: false as const, error: 'angles' as const };
    reading[key] = value;
  }
  if (reading.knee === null && reading.back === null && reading.arm === null && reading.hip === null) {
    return { ok: false as const, error: 'angles' as const };
  }
  return { ok: true as const, reading };
}
```

```ts
export type LinkDecision =
  | { action: 'login'; userId: number }
  | { action: 'attach'; userId: number }
  | { action: 'attach-email'; userId: number }
  | { action: 'create' }
  | { action: 'taken' };

export function decideLink(input: {
  sessionUserId: number | null;
  identityUserId: number | null;
  emailUserId: number | null;
}): LinkDecision {
  if (input.identityUserId !== null) {
    if (input.sessionUserId !== null && input.sessionUserId !== input.identityUserId) return { action: 'taken' };
    return { action: 'login', userId: input.identityUserId };
  }
  if (input.sessionUserId !== null) return { action: 'attach', userId: input.sessionUserId };
  if (input.emailUserId !== null) return { action: 'attach-email', userId: input.emailUserId };
  return { action: 'create' };
}
```

```ts
import { createHash, createHmac, timingSafeEqual } from 'node:crypto';

export function verifyTelegramLogin(fields: Record<string, string>, botToken: string, nowSec: number): boolean {
  const hash = fields.hash ?? '';
  const authDate = Number(fields.auth_date);
  if (!hash || !fields.id || !Number.isFinite(authDate)) return false;
  if (nowSec - authDate > 86400 || authDate > nowSec + 30) return false;
  const data = Object.keys(fields)
    .filter((key) => key !== 'hash')
    .sort()
    .map((key) => `${key}=${fields[key]}`)
    .join('\n');
  const secret = createHash('sha256').update(botToken).digest();
  const digest = createHmac('sha256', secret).update(data).digest();
  const given = Buffer.from(hash, 'hex');
  if (given.length !== digest.length) return false;
  return timingSafeEqual(given, digest);
}

export function telegramName(fields: { first_name?: string; last_name?: string; username?: string }): string {
  const full = [fields.first_name, fields.last_name].filter(Boolean).join(' ').trim();
  return (full || fields.username || 'Велосипедист').slice(0, 80);
}
```

```ts
export function apiUrl(): string {
  return (import.meta.env.PUBLIC_API_URL || '').replace(/\/$/, '');
}

export async function cabinetFetch(path: string, init: RequestInit = {}): Promise<Response> {
  const headers = new Headers(init.headers);
  headers.set('X-BikeTrue', '1');
  if (init.body) headers.set('Content-Type', 'application/json');
  return fetch(`${apiUrl()}${path}`, { ...init, headers, credentials: 'include' });
}
```
